```jsx
<Header 
  topbarText="Sản phẩm chính hãng - Xuất VAT đầy đủ"
  searchPlaceholder="Bạn tìm gì hôm nay? (Nồi chiên, Gaabor, Hunonic...)"

  onSearch={(q) => console.log(q)}
/>
```
